' Base64 encoding function for VBA
Function EncodeBase64(bytes() As Byte) As String
    Dim xmlObj As Object
    Set xmlObj = CreateObject("MSXML2.DOMDocument.6.0")
    Dim elem As Object
    Set elem = xmlObj.createElement("b64")
    elem.DataType = "bin.base64"
    elem.nodeTypedValue = bytes
    EncodeBase64 = elem.Text
    ' Remove any line breaks from the encoded string
    EncodeBase64 = Replace(EncodeBase64, vbCrLf, "")
End Function
